cmake_minimum_required(VERSION 3.20)
project(lumen_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SOURCES src/text_stats.c src/fuzzy.cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  enable_language(ASM)
  list(APPEND SOURCES src/count_newlines_x64.S)
  set(LUMEN_ASM 1)
else()
  list(APPEND SOURCES src/count_newlines_portable.c)
  set(LUMEN_ASM 0)
endif()
add_library(lumen_core SHARED ${SOURCES})
target_compile_definitions(lumen_core PRIVATE LUMEN_EXPORTS LUMEN_HAS_ASM=${LUMEN_ASM})
if(MSVC)
  target_compile_options(lumen_core PRIVATE /W4)
else()
  target_compile_options(lumen_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(lumen_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
